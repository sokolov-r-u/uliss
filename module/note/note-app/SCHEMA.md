The `note` schema after every Flyway migration, grouped by area. Generated by
`NoteSchemaSnapshotIntegrationTest` from `pg_dump`; do not edit by hand.
Update after a new migration: `./gradlew :note:updateSchemaSnapshot`.

## Notes

User notes and their posters.

### `notes`

```sql
CREATE TABLE note.notes (
    id uuid NOT NULL,
    user_id uuid NOT NULL,
    content text,
    status character varying(16) DEFAULT 'READY'::character varying NOT NULL,
    created_at timestamp with time zone NOT NULL,
    updated_at timestamp with time zone NOT NULL,
    version bigint,
    created_by character varying(64) NOT NULL,
    updated_by character varying(64) NOT NULL,
    source character varying(16) DEFAULT 'MANUAL'::character varying NOT NULL,
    title character varying(50),
    parent_note_id uuid,
    icon character varying(32),
    pinned_at timestamp with time zone,
    idempotency_key uuid,
    request_fingerprint bytea,
    CONSTRAINT chk_notes_not_own_parent CHECK ((parent_note_id <> id)),
    CONSTRAINT notes_source_check CHECK (((source)::text = ANY ((ARRAY['MANUAL'::character varying, 'CHAT_SUMMARY'::character varying])::text[]))),
    CONSTRAINT notes_status_check CHECK (((status)::text = ANY ((ARRAY['GENERATING'::character varying, 'READY'::character varying, 'FAILED'::character varying])::text[])))
);

ALTER TABLE ONLY note.notes
    ADD CONSTRAINT notes_pkey PRIMARY KEY (id);

ALTER TABLE ONLY note.notes
    ADD CONSTRAINT uq_notes_id_user_id UNIQUE (id, user_id);

ALTER TABLE ONLY note.notes
    ADD CONSTRAINT uq_notes_user_idempotency_key UNIQUE (user_id, idempotency_key);

CREATE INDEX idx_notes_parent_note_id ON note.notes USING btree (parent_note_id);

CREATE INDEX idx_notes_user_id ON note.notes USING btree (user_id);

CREATE INDEX idx_notes_user_pinned ON note.notes USING btree (user_id, pinned_at DESC, id DESC) WHERE (pinned_at IS NOT NULL);

CREATE INDEX idx_notes_user_updated_id ON note.notes USING btree (user_id, updated_at, id);

ALTER TABLE ONLY note.notes
    ADD CONSTRAINT fk_notes_parent_note FOREIGN KEY (parent_note_id, user_id) REFERENCES note.notes(id, user_id);
```

## Constellations

A user's tree of constellations and the notes placed in them.

### `constellations`

```sql
CREATE TABLE note.constellations (
    id uuid NOT NULL,
    user_id uuid NOT NULL,
    parent_id uuid,
    name character varying(40) NOT NULL,
    hue smallint,
    lightness character varying(8),
    created_at timestamp with time zone NOT NULL,
    updated_at timestamp with time zone NOT NULL,
    version bigint,
    created_by character varying(64) NOT NULL,
    updated_by character varying(64) NOT NULL,
    CONSTRAINT chk_constellations_colour CHECK ((((parent_id IS NULL) AND (hue IS NOT NULL) AND ((hue >= 0) AND (hue <= 359)) AND (lightness IS NULL)) OR ((parent_id IS NOT NULL) AND (hue IS NULL) AND (lightness IS NOT NULL) AND ((lightness)::text = ANY ((ARRAY['DEEP'::character varying, 'BASE'::character varying, 'LIGHT'::character varying, 'PALE'::character varying])::text[]))))),
    CONSTRAINT chk_constellations_not_own_parent CHECK ((parent_id <> id))
);

ALTER TABLE ONLY note.constellations
    ADD CONSTRAINT constellations_pkey PRIMARY KEY (id);

ALTER TABLE ONLY note.constellations
    ADD CONSTRAINT uq_constellations_id_user_id UNIQUE (id, user_id);

CREATE INDEX idx_constellations_parent_id ON note.constellations USING btree (parent_id);

CREATE UNIQUE INDEX uq_constellations_user_parent_name ON note.constellations USING btree (user_id, parent_id, lower((name)::text)) NULLS NOT DISTINCT;
```

## Outbox

Durable, retried background work (summaries, indexing, profile refresh).

### `outbox_event`

```sql

    id uuid NOT NULL,
    type character varying(32) NOT NULL,
    payload jsonb NOT NULL,
    status character varying(16) NOT NULL,
    attempts integer DEFAULT 0 NOT NULL,
    next_attempt_at timestamp with time zone NOT NULL,
    last_error text,
    created_at timestamp with time zone NOT NULL,
    updated_at timestamp with time zone NOT NULL,
    version bigint,
    lease_id uuid,

    CONSTRAINT outbox_event_status_check CHECK (((status)::text = ANY ((ARRAY['PENDING'::character varying, 'PROCESSING'::character varying, 'COMPLETED'::character varying, 'FAILED'::character varying])::text[]))),
    CONSTRAINT outbox_event_type_check CHECK (((type)::text = ANY ((ARRAY['NOTE_SUMMARY_REQUESTED'::character varying, 'NOTE_SUMMARY_UPDATE_REQUESTED'::character varying, 'NOTE_INDEX_REQUESTED'::character varying, 'NOTE_RESUMMARIZE_REQUESTED'::character varying, 'USER_PROFILE_REFRESH_REQUESTED'::character varying])::text[])))
);

ALTER TABLE ONLY note.outbox_event
    ADD CONSTRAINT outbox_event_pkey PRIMARY KEY (id);

CREATE INDEX idx_outbox_event_status_next_attempt ON note.outbox_event USING btree (status, next_attempt_at);

CREATE UNIQUE INDEX uq_outbox_event_pending_dedup_key ON note.outbox_event USING btree (dedup_key) WHERE (((status)::text = 'PENDING'::text) AND (dedup_key IS NOT NULL));
```
